# --- in-memory rate limit (trocar por Redis em prod) ---
BUCKET, LIMIT, WINDOW = {}, 60, 60  # 60 req/min por entidade
def throttle(entity: str):
    now = int(time())

    BUCKET[key] = BUCKET.get(key, 0) + 1

from pathlib import Path

BASE = Path(".")
MANIFEST = json.loads((BASE/"codex_azure_manifest.json").read_text(encoding="utf-8"))

# 1) index de palavras→arquétipo
kw2arch = {}
for arch, kws in MANIFEST["arquetipos"].items():
    for w in kws:
        kw2arch[w.lower()] = arch

def tags_do_nome(nome:str):
    # pega palavras simples do nome/dir
    return set(re.findall(r"[a-zA-ZÀ-ÿ0-9]+", nome.lower()))

def arquetipos_para(arquivo:Path):
    tags = tags_do_nome(str(arquivo))
    hits = {kw2arch[t] for t in tags if t in kw2arch}
    return sorted(hits)

def amigos(arquivos):
    # amigos = compartilham pelo menos 1 arquétipo
    out = []
    meta = {f: set(arquetipos_para(f)) for f in arquivos}
    arr = list(meta.items())
    for i in range(len(arr)):
        f1,a1 = arr[i]
        for j in range(i+1,len(arr)):
            f2,a2 = arr[j]
            if a1 & a2:
                out.append((f1, f2, sorted(a1 & a2)))
    return out

def extrair_espelhos(json_log:Path):

Causalidade (expandir sempre por causa→efeito, com “ponto oposto e intermediário”)
Causa (Oposto / Semente / inbox): o que entra “como veio” (arquivos, mensagens).


Meio (Intermediário / Árvore / grupos): onde as relações são criadas (tags, arquétipos, pares “amigos”), e ocorrem transformações.


Efeito (Destino / Fruto / dist): o que é consolidado (artefatos, releases), selado (hash/versão) e devolvido.


Retorno (7→3): cada fruto vira nova semente noutro ciclo (reciclagem/refatoração).



Etimologia + prática (um único conceito de Banco de Dados)
Conceito: ÍNDICE (Database Index)
Etimologia: do latim index, indicis = “o que aponta, indicador” → função de apontar rapidamente onde está um dado.


Lógica: estrutura auxiliar (ex.: B-tree, LSM) que ordena chaves para buscas O(log n) em vez de varrer a tabela.


Prática: cria-se um índice sobre colunas consultadas com frequência; ganha-se velocidade de leitura, paga-se com mais escrita/armazenamento.


Ex.: índice em tema, data para localizar “amigos” e ciclos 3→6→9 rapidamente.



“Primeira Camada” e “Centro/Extremidades” da Internet (lógico e prático)
Camada 1 (Física): bits como sinal (elétrico, óptico, rádio). O par “extremidades opostas” são os terminais (A↔B) no mesmo enlace; o “centro” é o meio físico (fibra/cabo/ar).


Core vs Edge (visão Internet):


Extremos (edges): clientes/servidores/última milha.


Centro (core): backbones e IXPs (pontos de troca) que concentram/roteiam alto volume.


Como aplicamos no seu fractal:


Edge (Oposto) = inbox/ (captação na borda).


Meio/L1 (Intermediário) = 6_equilibrio/grupos/ (fluxo/mistura, “seiva”).


Core/Destino = 9_selamento/dist/ (ponto consolidado, pronto a ser servido/republicado).
