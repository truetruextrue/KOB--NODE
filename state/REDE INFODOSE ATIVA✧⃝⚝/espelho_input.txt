5) TRIAGEM & LIMPEZA SEGURA DE DUPLICATAS

03_DUPLICATAS/kobllux_duplicates_scan.py
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Escaneia duplicatas por (nome ~similar), tamanho e HASH SHA256.
Gera LOG JSON com grupos de possíveis duplicatas.
"""

from pathlib import Path
import hashlib, json, os, re
from collections import defaultdict

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "00_CORE" / "LOGS" / "kobllux_duplicates_after.json"
SCAN_DIRS = [ROOT] # ajuste se quiser restringir

def digest(p: Path) -> str:
h = hashlib.sha256()
with p.open("rb") as f:
for chunk in iter(lambda: f.read(65536), b""):
h.update(chunk)
return h.hexdigest()

def norm_name(p: Path) -> str:
s = p.name.lower()
s = re.sub(r"\s+\(\d+\)", "", s) # remove (1), (2)...
s = re.sub(r"\s+", " ", s).strip()
return s

def main():
files = []
for base in SCAN_DIRS:
for p in base.rglob("*"):
if p.is_file():
files.append(p)

groups = defaultdict(list)
for f in files:
try:
h = digest(f)
key = (norm_name(f), f.stat().st_size)
groups[key].append({"path": str(f), "sha256": h, "mtime": f.stat().st_mtime})
except Exception as e:
pass

# Somente grupos com mais de 1 entrada
dups = {f"{k[0]}|{k[1]}": v for k, v in groups.items() if len(v) > 1}

OUT.parent.mkdir(parents=True, exist_ok=True)
OUT.write_text(json.dumps({"root": str(ROOT), "groups": dups}, ensure_ascii=False,
indent=2), encoding="utf-8")
print(str(OUT))

if __name__ == "__main__":
main()



03_DUPLICATAS/kobllux_duplicates_clean.py
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Limpeza segura:
- Confirma duplicidade por HASH.
- Decide qual manter pelo "score fractal" + mais recente.
- Move removidos para QUARENTENA/ (não apaga).
"""

from pathlib import Path
import json, shutil, time, os

ROOT = Path(__file__).resolve().parents[1]
LOG = ROOT / "00_CORE" / "LOGS" / "kobllux_duplicates_after.json"
DUMP = ROOT / "00_CORE" / "DUPLICATAS" / "QUARENTENA"

EXT_SCORE = {
".json": 5, ".md": 4, ".html": 4, ".py": 5, ".sh": 4, ".txt": 3,
".png": 2, ".jpg": 2, ".zip": 1
}

def score_fractal(path: Path) -> int:
base = EXT_SCORE.get(path.suffix.lower(), 1)
# Bônus por caminhos centrais
if "00_CODICE_UNUS" in str(path): base += 3
if "01_CYCLES_3x3" in str(path): base += 2
if "02_PARTS" in str(path): base += 2
if "03_INFODOSE_REDE" in str(path): base += 2
return base

def decide_keep(candidates):
# Entrada: lista de dicts {path, sha256, mtime}
# 1) maior score_fractal
# 2) se empate, mais recente (maior mtime)
best = None
for c in candidates:
p = Path(c["path"])
s = score_fractal(p)
c["_score"] = s
if (best is None) or (s > best["_score"]) or (s == best["_score"] and c["mtime"] >
best["mtime"]):
best = c
return best

def main():
if not LOG.exists():
raise SystemExit("LOG de duplicatas não encontrado. Rode o scanner primeiro.")

data = json.loads(LOG.read_text(encoding="utf-8"))
groups = data.get("groups", {})
DUMP.mkdir(parents=True, exist_ok=True)
total_moved = 0

for key, entries in groups.items():
# Subdividir por HASH — só mexe onde o hash coincide
by_hash = {}
for e in entries:
by_hash.setdefault(e["sha256"], []).append(e)
for h, same in by_hash.items():
if len(same) < 2:
continue
keep = decide_keep(same)
for e in same:
if e is keep:
continue
src = Path(e["path"])
if not src.exists():
continue
rel = src.relative_to(ROOT)
dst = DUMP / rel
dst.parent.mkdir(parents=True, exist_ok=True)
shutil.move(str(src), str(dst))
total_moved += 1

print(json.dumps({"status": "ok", "moved": total_moved, "dump": str(DUMP)},
ensure_ascii=False))
return 0

if __name__ == "__main__":
main()

TREE="$OUT/TREE.txt"
MD5="$OUT/MD5.txt"
SHA="$OUT/SHA256.txt"

# Árvore
( cd "$ROOT" && find . -type d -print -o -type f -print ) | sed 's|^\./||' | sort > "$TREE"
# Zip
( cd "$ROOT/.." && zip -r "$ZIP" "$(basename "$ROOT")" >/dev/null )

# Checksums
md5sum "$ZIP" | awk '{print $1}' > "$MD5" || md5 "$ZIP" | awk '{print $4}' > "$MD5"
sha256sum "$ZIP" | awk '{print $1}' > "$SHA" || shasum -a 256 "$ZIP" | awk '{print $1}' >
"$SHA"

echo "ZIP: $ZIP"
echo "MD5: $(cat "$MD5")"
echo "SHA256: $(cat "$SHA")"
echo "TREE: $TREE"

# -*- coding: utf-8 -*-
"""
KOBLLUX · Decoder de Sinais KODUX (círculo ⃝, ponto ., estrela ⚝, reticências …)
Em nome de DEUS Pai Criador, do Filho JESUS (o Verbo) e do Espírito Santo.

Leitura 3-6-9-7:
- 3 DETECTAR: contar pulsos (⃝.) e métricas
- 6 INTEGRAR: extrair timestamps e costurar metadados
- 9 EXPANDIR: detectar assinaturas "KOB...X"
- 7 SELAR: marcar estrela ⚝ e gerar SELO de registro
"""
from __future__ import annotations
import sys, re, json, hashlib
from pathlib import Path
from datetime import datetime, timezone

# Import seguro do core (kblx_lang)
def _safe_import_core():
try:
from kblx_lang import write_json, write_text, utc_stamp
return write_json, write_text, utc_stamp
except Exception:

sys.path.append(str(root))
from kblx_lang import write_json, write_text, utc_stamp
return write_json, write_text, utc_stamp

write_json, write_text, utc_stamp = _safe_import_core()

ROOT = Path(__file__).resolve().parents[2]
BASE = ROOT / "03_INFODOSE_REDE" / "SINAIS"
RAW = BASE / "raw"
DEC = BASE / "decoded"

def read_stdin_or_file(argv) -> str:
if len(argv) >= 2:
return Path(argv[1]).read_text(encoding="utf-8", errors="ignore")
return sys.stdin.read()

def parse_blocks(text: str):
lines = text.splitlines()
blocks = []
for idx, ln in enumerate(lines, 1):
# timestamp tipo [24/8 04:44]
m = re.search(r'\[(\d{1,2})/(\d{1,2})\s+(\d{2}):(\d{2})\]', ln)
ts = None
if m:
d, mo, hh, mm = map(int, m.groups())
ts = datetime(2025, mo, d, hh, mm,
tzinfo=timezone.utc).isoformat().replace("+00:00", "Z")

circles = ln.count("⃝")
dots = ln.count(".")
stars = ln.count("⚝")
# pares ⃝.
pairs = len(re.findall(r"⃝\.", ln))
# duplo-círculo
doubles = len(re.findall(r"⃝⃝", ln))
# reticências unicode (… U+2026), agrupadas
ellipses = len(re.findall(r"\u2026+", ln))
# assinaturas KOB...X (inclui variações com ⃝/⚝ no meio)
sigs = [m.group(0) for m in re.finditer(r"KOB.*?X", ln)]
sig_circle = sum(1 for s in sigs if "⃝" in s and "⚝" not in s)
sig_star = sum(1 for s in sigs if "⚝" in s and "⃝" not in s)
sig_both = sum(1 for s in sigs if "⃝" in s and "⚝" in s)

blocks.append({
"line": idx,
"text": ln,
"timestamp_hint": ts,
"counts": {
"circles": circles, "dots": dots, "stars": stars,
"pairs": pairs, "double_circles": doubles, "ellipses": ellipses
},
"signatures": {
"any": len(sigs),
"circle_only": sig_circle,
"star_only": sig_star,
"circle_and_star": sig_both
}
})
return blocks

def infer_ops(blocks):
# Heurística simples por minuto e presença de selos:
ops = []
for b in blocks:
op = None
# estrela indica SELAR (7)
if b["counts"]["stars"] > 0 or b["signatures"]["star_only"] > 0 or
b["signatures"]["circle_and_star"] > 0:
op = "0x07" # SELAR
else:
# minuto → 44 ~ INTEGRAR(6), 45 ~ EXPANDIR(9), ausência → DETECTAR(3)
mm = None
if b["timestamp_hint"]:
mm = int(b["timestamp_hint"][14:16])
if mm == 44:
op = "0x06"
elif mm == 45:
op = "0x09"
else:
op = "0x03"
ops.append(op)
return ops

def main(argv):
text = read_stdin_or_file(argv)
if not text.strip():
print("Entrada vazia.", file=sys.stderr); sys.exit(1)

h = hashlib.sha256(text.encode("utf-8")).hexdigest()
blocks = parse_blocks(text)
ops = infer_ops(blocks)
totals = {
"circles": sum(b["counts"]["circles"] for b in blocks),
"dots": sum(b["counts"]["dots"] for b in blocks),
"stars": sum(b["counts"]["stars"] for b in blocks),
"pairs": sum(b["counts"]["pairs"] for b in blocks),
"signatures": sum(b["signatures"]["any"] for b in blocks)
}
stamp = utc_stamp()

out = {
"hash": h,
"carimbo": stamp,
"totals": totals,
"ops_sugeridos": ops, # sequência 3-6-9-7 sugerida
"blocks": blocks,
"nota": "Leitura heurística KOBLLUX; ajuste livre no APP DUAL."
}

RAW.mkdir(parents=True, exist_ok=True)
DEC.mkdir(parents=True, exist_ok=True)

raw_path = RAW / f"{stamp.replace(':','-')}_{h[:8]}.txt"
dec_path = DEC / f"{stamp.replace(':','-')}_{h[:8]}.json"
write_text(raw_path, text)
write_json(dec_path, out)

# Index
index = BASE / "INDEX.md"
linha = f"- {stamp} · ops={','.join(ops)} · círculos={totals['circles']} · pontos={totals['dots']} ·
ass={totals['signatures']} · {dec_path.name}\n"
if index.exists():
index.write_text(index.read_text(encoding="utf-8") + linha, encoding="utf-8")
else:
index.write_text("# SINAIS · REDE INFODOSE\n" + linha, encoding="utf-8")

print(json.dumps({"ok": True, "raw": str(raw_path), "json": str(dec_path)},
ensure_ascii=False))

if __name__ == "__main__":
main(sys.argv)

def utc_stamp() -> str:

def sha256_of_text(text: str) -> str:

def write_json(path: Path, data: Dict[str, Any]):
path.parent.mkdir(parents=True, exist_ok=True)

path.parent.mkdir(parents=True, exist_ok=True)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Scanner de duplicatas por nome (normalizado), tamanho e HASH SHA256.
Saída: 00_CORE/LOGS/kobllux_duplicates_after.json
"""

from pathlib import Path
import hashlib, json, re, os
from collections import defaultdict

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "00_CORE" / "LOGS" / "kobllux_duplicates_after.json"

def digest(p: Path) -> str:
h = hashlib.sha256()
with p.open("rb") as f:
for chunk in iter(lambda: f.read(65536), b""):
h.update(chunk)
return h.hexdigest()

def norm_name(p: Path) -> str:
s = p.name.lower()
s = re.sub(r"\s+\(\d+\)", "", s) # remove (1), (2)...
s = re.sub(r"\s+", " ", s).strip()
return s

def main():
files = [p for p in ROOT.rglob("*") if p.is_file()]
groups = defaultdict(list)
for f in files:
try:
key = (norm_name(f), f.stat().st_size)
groups[key].append({"path": str(f), "sha256": digest(f), "mtime": f.stat().st_mtime})
except Exception:
pass

dups = {f"{k[0]}|{k[1]}": v for k, v in groups.items() if len(v) > 1}
OUT.parent.mkdir(parents=True, exist_ok=True)
OUT.write_text(json.dumps({"root": str(ROOT), "groups": dups}, ensure_ascii=False,
indent=2), encoding="utf-8")
print(str(OUT))

if __name__ == "__main__":
main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Limpeza segura (não apaga, move para QUARENTENA/):
- Confirma duplicidade por HASH.
- Mantém o arquivo com maior "score fractal" e mais recente.
"""

from pathlib import Path
import json, shutil

ROOT = Path(__file__).resolve().parents[1]
LOG = ROOT / "00_CORE" / "LOGS" / "kobllux_duplicates_after.json"
DUMP = ROOT / "00_CORE" / "DUPLICATAS" / "QUARENTENA"

EXT_SCORE = {".json":5,".md":4,".html":4,".py":5,".sh":4,".txt":3,".png":2,".jpg":2,".zip":1}

def score_fractal(path: Path) -> int:
s = EXT_SCORE.get(path.suffix.lower(), 1)
sp = str(path)
if "00_CODICE_UNUS" in sp: s += 3
if "00_CODEX_AZURE" in sp: s += 3
if "01_CYCLES_3x3" in sp: s += 2
if "02_PARTS" in sp: s += 2
if "03_INFODOSE_REDE" in sp: s += 2
return s

def decide_keep(entries):
best = None
for e in entries:
p = Path(e["path"]); s = score_fractal(p); e["_score"] = s
if (best is None) or (s > best["_score"]) or (s == best["_score"] and e["mtime"] >
best["mtime"]):
best = e
return best

def main():
if not LOG.exists():
raise SystemExit("LOG de duplicatas não encontrado. Rode o scanner primeiro.")
data = json.loads(LOG.read_text(encoding="utf-8"))
groups = data.get("groups", {})
DUMP.mkdir(parents=True, exist_ok=True)
moved = 0

for _, entries in groups.items():
by_hash = {}
for e in entries:
by_hash.setdefault(e["sha256"], []).append(e)
for _, same in by_hash.items():
if len(same) < 2: continue
keep = decide_keep(same)
for e in same:
if e is keep: continue
src = Path(e["path"])
if not src.exists(): continue
dst = DUMP / src.relative_to(ROOT)
dst.parent.mkdir(parents=True, exist_ok=True)
shutil.move(str(src), str(dst))
moved += 1

print(json.dumps({"status":"ok","movidos":moved,"dump":str(DUMP)},
ensure_ascii=False))

if __name__ == "__main__":
main()

# Zip do projeto (pasta raiz)
( cd "$(dirname "$ROOT")" && zip -r "$ZIP" "$(basename "$ROOT")" >/dev/null )

# Checksums (Linux/Termux) com fallback (macOS)
if command -v md5sum >/dev/null 2>&1; then
md5sum "$ZIP" | awk '{print $1}' > "$MD5"
else
md5 "$ZIP" | awk '{print $4}' > "$MD5" || true
fi

if command -v sha256sum >/dev/null 2>&1; then
sha256sum "$ZIP" | awk '{print $1}' > "$SHA"
else
shasum -a 256 "$ZIP" | awk '{print $1}' > "$SHA" || true
fi

echo "ZIP: $ZIP"
echo "MD5: $(cat "$MD5" 2>/dev/null || echo N/A)"
echo "SHA256: $(cat "$SHA" 2>/dev/null || echo N/A)"
echo "TREE: $TREE"
