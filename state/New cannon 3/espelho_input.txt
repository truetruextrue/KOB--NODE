class ColetorAST(ast.NodeVisitor):
def __init__(self) -> None:
self.imports: List[str] = []
self.from_imports: List[str] = []
self.funcs: List[Dict[str, Any]] = []
self.classes: List[Dict[str, Any]] = []
self.assigns: List[str] = []
self.ifs: int = 0
self.whiles: int = 0
self.fors: int = 0
self.returns: int = 0
self.raises: int = 0
self.trys: int = 0
self.withs: int = 0
self.comps: Dict[str, int] = {"list":0, "dict":0, "set":0, "gen":0}
self.lambdas: int = 0
self.decorators: int = 0
self.yields: int = 0

def visit_Import(self, node: ast.Import) -> None:
for alias in node.names:
self.imports.append(alias.name)
self.generic_visit(node)

def visit_ImportFrom(self, node: ast.ImportFrom) -> None:
base = node.module or ""
names = ", ".join(a.name for a in node.names)
self.from_imports.append(f"{base}: {names}")
self.generic_visit(node)

def visit_FunctionDef(self, node: ast.FunctionDef) -> None:
args = [a.arg for a in node.args.args]
if node.args.vararg: args.append("*" + node.args.vararg.arg)
if node.args.kwarg: args.append("**" + node.args.kwarg.arg)
self.funcs.append({"name": node.name, "args": args, "decorators":
len(node.decorator_list)})
self.decorators += len(node.decorator_list)
self.generic_visit(node)

def visit_AsyncFunctionDef(self, node: ast.AsyncFunctionDef) -> None:
self.visit_FunctionDef(node)

def visit_ClassDef(self, node: ast.ClassDef) -> None:
methods = [n.name for n in node.body if isinstance(n, (ast.FunctionDef,
ast.AsyncFunctionDef))]
bases = [ast.unparse(b) if hasattr(ast, "unparse") else getattr(b, 'id', '?') for b in
node.bases]
self.classes.append({"name": node.name, "methods": methods, "bases": bases,
"decorators": len(node.decorator_list)})
self.decorators += len(node.decorator_list)
self.generic_visit(node)

def visit_Assign(self, node: ast.Assign) -> None:
for t in node.targets:
if isinstance(t, ast.Name):
self.assigns.append(t.id)
self.generic_visit(node)

def visit_If(self, node: ast.If) -> None:
self.ifs += 1
self.generic_visit(node)

def visit_For(self, node: ast.For) -> None:
self.fors += 1
self.generic_visit(node)

def visit_While(self, node: ast.While) -> None:
self.whiles += 1
self.generic_visit(node)

def visit_Return(self, node: ast.Return) -> None:
self.returns += 1
self.generic_visit(node)

def visit_Raise(self, node: ast.Raise) -> None:
self.raises += 1
self.generic_visit(node)

def visit_Try(self, node: ast.Try) -> None:
self.trys += 1
self.generic_visit(node)

def visit_With(self, node: ast.With) -> None:
self.withs += 1
self.generic_visit(node)

def visit_Lambda(self, node: ast.Lambda) -> None:
self.lambdas += 1
self.generic_visit(node)

def visit_Yield(self, node: ast.Yield) -> None:
self.yields += 1
self.generic_visit(node)

def visit_ListComp(self, node: ast.ListComp) -> None:
self.comps["list"] += 1
self.generic_visit(node)
def visit_DictComp(self, node: ast.DictComp) -> None:
self.comps["dict"] += 1
self.generic_visit(node)

def visit_SetComp(self, node: ast.SetComp) -> None:
self.comps["set"] += 1
self.generic_visit(node)

def visit_GeneratorExp(self, node: ast.GeneratorExp) -> None:
self.comps["gen"] += 1
self.generic_visit(node)

class ColetorAST(ast.NodeVisitor):
def __init__(self) -> None:
self.imports: List[str] = []
self.from_imports: List[str] = []
self.funcs: List[Dict[str, Any]] = []
self.classes: List[Dict[str, Any]] = []
self.assigns: List[str] = []
self.ifs: int = 0
self.whiles: int = 0
self.fors: int = 0
self.returns: int = 0

def visit_Import(self, node: ast.Import) -> None:
for alias in node.names:
self.imports.append(alias.name)
self.generic_visit(node)

def visit_ImportFrom(self, node: ast.ImportFrom) -> None:
base = node.module or ""
names = ", ".join(a.name for a in node.names)
self.from_imports.append(f"{base}: {names}")
self.generic_visit(node)

def visit_FunctionDef(self, node: ast.FunctionDef) -> None:
args = [a.arg for a in node.args.args]
self.funcs.append({"name": node.name, "args": args})
self.generic_visit(node)

def visit_ClassDef(self, node: ast.ClassDef) -> None:
self.classes.append({"name": node.name})
self.generic_visit(node)
def visit_Assign(self, node: ast.Assign) -> None:
for t in node.targets:
if isinstance(t, ast.Name):
self.assigns.append(t.id)
self.generic_visit(node)

def visit_If(self, node: ast.If) -> None:
self.ifs += 1
self.generic_visit(node)

def visit_For(self, node: ast.For) -> None:
self.fors += 1
self.generic_visit(node)

def visit_While(self, node: ast.While) -> None:
self.whiles += 1
self.generic_visit(node)

def visit_Return(self, node: ast.Return) -> None:
self.returns += 1
self.generic_visit(node)

'p.add_argument("--theme", choices=["default","noite","luz","cosmico"], default="default",

'p.add_argument("--auto-theme", action="store_true", help="Escolher tema
automaticamente com base no código analisado.")'
)

# 2) Inject analysis function to pick theme based on code semantics
injection_point = 'def explicar_codigo_fonte(source: str, titulo: Optional[str] = None):'
theme_func = r'''
def _analisar_tema_semantico(source: str) -> str:
"""
Heurística de seleção de tema:
- "cosmico": se detectar 3-6-9 (lista ou sequência) ou uso de 'yield'/'async'
- "noite": se houver muitos try/except (>=2)
- "luz": se houver muitas estruturas (funções+classes >= 6) e >=2 imports
- "default": caso contrário
"""
import ast, re
class V(ast.NodeVisitor):
def __init__(self):
self.trys = 0
self.funcs = 0
self.classes = 0
self.imports = 0
self.yields = 0
self.async_funcs = 0
self.fors = 0
self.whiles = 0
def visit_Try(self, node): self.trys += 1; self.generic_visit(node)
def visit_FunctionDef(self, node): self.funcs += 1; self.generic_visit(node)
def visit_AsyncFunctionDef(self, node): self.async_funcs += 1; self.generic_visit(node)
def visit_ClassDef(self, node): self.classes += 1; self.generic_visit(node)
def visit_Import(self, node): self.imports += 1; self.generic_visit(node)
def visit_ImportFrom(self, node): self.imports += 1; self.generic_visit(node)
def visit_Yield(self, node): self.yields += 1; self.generic_visit(node)
def visit_For(self, node): self.fors += 1; self.generic_visit(node)
def visit_While(self, node): self.whiles += 1; self.generic_visit(node)
try:
tree = ast.parse(source)
except SyntaxError:
return "default"
v = V(); v.visit(tree)
# padrões 3-6-9 explícitos
cosmico_pattern = re.compile(r"(3\s*[-,]\s*6\s*[-,]\s*9)|(\[\s*3\s*,\s*6\s*,\s*9\s*\])")
if cosmico_pattern.search(source) or v.yields > 0 or v.async_funcs > 0:
return "cosmico"
if v.trys >= 2:
return "noite"
if (v.funcs + v.classes) >= 6 and v.imports >= 2:
return "luz"
return "default"
'''
code = code.replace(injection_point, theme_func + "\n" + injection_point)

# 3) In main(), after applying explicit theme / no-color, add auto-theme logic for
