# ~/KOBLLUX/scripts/bash/kobllux_ingest.sh
#!/data/data/com.termux/files/usr/bin/bash
set -euo pipefail

ROOT=~/KOBLLUX
INBOX="$ROOT/inbox"
OUT_TXT="$ROOT/out/txt"
OUT_OBJ="$ROOT/out/obj"
OUT_JSON="$ROOT/out/json"
CYCLE1="$ROOT/cycle/1_DETECTAR"
CYCLE2="$ROOT/cycle/2_INTEGRAR"
CYCLE3="$ROOT/cycle/3_EXPANDIR"
CYCLE7="$ROOT/cycle/7_SELAR"
ARCH="$ROOT/archive/originais/$(date +%Y%m%d)"
LOG="$ROOT/logs/run_$(date +%Y%m%d_%H%M).log"

mkdir -p "$OUT_TXT" "$OUT_OBJ" "$OUT_JSON" "$CYCLE1" "$CYCLE2" "$CYCLE3" "$CYCLE7" "$ARCH" "$ROOT/logs"

DOC="${1:-}"
if [ -z "$DOC" ]; then
  DOC="$(ls -1 "$INBOX" | head -n1 || true)"
fi
[ -n "$DOC" ] || { echo "Nada em inbox/"; exit 0; }

echo "[UNO] DETECTAR $DOC" | tee -a "$LOG"
python $ROOT/python/lib/kobllux_theme.py 2>/dev/null || true
echo "{\"arquivo\":\"$DOC\",\"quando\":\"$(date -Iseconds)\"}" > "$CYCLE1/${DOC%.*}.meta.json"

echo "[DUO] INTEGRAR (narrativa base)" | tee -a "$LOG"
python $ROOT/python/documentos/verbo_agua_codigo.py > "$OUT_TXT/${DOC%.*}.narra.txt"
echo "{\"arquivo\":\"$DOC\",\"decisoes\":[\"narrativa_base\"]}" > "$CYCLE2/${DOC%.*}.manifest.json"

echo "[TRINITY] EXPANDIR (fractal + json)" | tee -a "$LOG"
python $ROOT/python/bridges/kobllux_narrativo_geom.py \
  --json --generate-fractal --level 2 \
  --out $OUT_OBJ/sierpinski_n2.obj \
  --out-mesh $OUT_OBJ/sierpinski_mesh_n2.obj \
  | tee $OUT_JSON/run_$(date +%Y%m%d_%H%M).json

echo "[SELO] mover original e registrar selo" | tee -a "$LOG"
mv "$INBOX/$DOC" "$ARCH/"
jq -n --arg doc "$DOC" --arg quando "$(date -Iseconds)" \
  '{selo:"VERDADE × INTEGRAR ÷ ∆ = ♾️",quando:$quando,doc:$doc}' \
  > "$CYCLE7/${DOC%.*}.selo.json"

echo "[FIM] Selado ∴" | tee -a "$LOG"

# -*- coding: utf-8 -*-
"""
veeb_runtime_gtts.py — Runtime TTS (gTTS) para KOBLLUX
- Lê configs/voices.json
- Salva MP3 em out/audio/AAAAMMDD/
- Emite evento JSON em data/events/
- Tenta tocar via termux-media-player (opcional)
"""
from __future__ import annotations
import json, hashlib, time, subprocess, re, os
from pathlib import Path
from datetime import datetime, timezone
from typing import Tuple, Dict, Any, Optional

try:
    from gtts import gTTS
    _HAS_GTTS = True
except Exception:
    _HAS_GTTS = False

ROOT = Path.home() / "KOBLLUX"
CFG_PATH = ROOT / "configs" / "voices.json"
OUT_AUDIO_ROOT = ROOT / "out" / "audio"
EVENTS_ROOT = ROOT / "data" / "events"

def _ensure_dir(p: Path) -> Path:
    p.mkdir(parents=True, exist_ok=True); return p

def _slug(s: str, maxlen: int = 48) -> str:
    s = re.sub(r"[^\w\d]+", "_", s.strip(), flags=re.UNICODE)
    s = re.sub(r"_+", "_", s).strip("_")
    return s[:maxlen] if s else "texto"

def _ymd_dir(root: Path) -> Path:
    return _ensure_dir(root / datetime.now(timezone.utc).astimezone().strftime("%Y%m%d"))

def _hashes(path: Path) -> Dict[str, str]:
    md5 = hashlib.md5(); sha = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            md5.update(chunk); sha.update(chunk)
    return {"md5": md5.hexdigest(), "sha256": sha.hexdigest()}

def _load_cfg(cfg_path: Path = CFG_PATH) -> Dict[str, Any]:
    if cfg_path.exists():
        return json.loads(cfg_path.read_text(encoding="utf-8"))
    return {
        "engine_priority": ["gtts"],
        "default_lang": "pt",
        "default_tld": "com.br",
        "archetype_voices": {"NOVA": {"engine": "gtts", "voice": "pt-BR-NOVA"}}
    }

def _termux_play(path: Path) -> bool:
    # Toca via termux-media-player (se instalado). Fallback: none.
    try:
        subprocess.run(["termux-media-player", "play", str(path)],
                       stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=False)
        return True
    except Exception:
        return False

def tts_gtts(texto: str, lang: str = "pt", tld: str = "com.br", outfile: Path = None) -> Path:
    if not _HAS_GTTS:
        raise RuntimeError("gTTS não disponível. Instale com: pip install gTTS")
    if outfile is None:
        raise ValueError("outfile obrigatório")
    tts = gTTS(text=texto, lang=lang, tld=tld)
    tts.save(str(outfile))
    return outfile

def speak(
    texto: str,
    archetype: str = "NOVA",
    engine_override: Optional[str] = None,
    cfg_path: Path = CFG_PATH,
    out_root: Path = OUT_AUDIO_ROOT,
    events_root: Path = EVENTS_ROOT,
    play: bool = True,
) -> Dict[str, Any]:
    """
    Gera áudio a partir de texto usando mapeamento de voices.json.
    Retorna um dicionário com metadata (path, hashes, archetype, engine, timestamp).
    """
    cfg = _load_cfg(cfg_path)
    arche = archetype.upper()
    voices = cfg.get("archetype_voices", {})
    if arche not in voices:
        # fallback para NOVA
        arche = "NOVA"
    v = voices.get(arche, {})
    engine = (engine_override or v.get("engine") or cfg.get("engine_priority", ["gtts"])[0]).lower()

    # Resolução de parâmetros gTTS
    lang = cfg.get("default_lang", "pt")
    tld = cfg.get("default_tld", "com.br")

    # Caminho de saída
    day_dir = _ymd_dir(out_root)
    stem = f"{datetime.now().strftime('%H%M%S')}_{_slug(arche)}_{_slug(texto)}"
    out_path = day_dir / f"{stem}.mp3"

    # Execução por engine
    if engine == "gtts":
        tts_gtts(texto, lang=lang, tld=tld, outfile=out_path)
    else:
        raise NotImplementedError(f"Engine não implementada neste runtime: {engine}")

    # Hashes + evento
    h = _hashes(out_path)
    event = {
        "evento": "tts_emitido",
        "quando_utc": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
        "arquetipo": arche,
        "engine": engine,
        "arquivo": str(out_path),
        "hashes": h,
        "bytes": out_path.stat().st_size,
        "axioma": "VERDADE × INTEGRAR ÷ ∆ = ♾️"
    }
    _ensure_dir(events_root)
    ev_path = events_root / f"{int(time.time())}_{_slug(arche)}.json"
    ev_path.write_text(json.dumps(event, ensure_ascii=False, indent=2), encoding="utf-8")

    # Tocar (opcional)
    if play:
        _termux_play(out_path)

    return event

if __name__ == "__main__":
    # Execução rápida (teste): python veeb_runtime_gtts.py "texto" HÓRUS
    import sys
    txt = sys.argv[1] if len(sys.argv) > 1 else "KOBLLUX ATIVAR — VERDADE × INTEGRAR ÷ ∆ = ♾️"
    arc = sys.argv[2] if len(sys.argv) > 2 else "NOVA"
    meta = speak(txt, arc, play=False)
    print(json.dumps(meta, ensure_ascii=False, indent=2))

# -*- coding: utf-8 -*-
"""
kobllux_cli.py — CLI simples para TTS do KOBLLUX usando voices.json
Exemplos:
  python python/tools/kobllux_cli.py --text "Glória" --archetype HÓRUS
  python python/tools/kobllux_cli.py --file ~/KOBLLUX/inbox/mensagem.txt --archetype NOVA
  python python/tools/kobllux_cli.py --text "Trinity" --no-play
"""
from __future__ import annotations
import argparse, json, sys
from pathlib import Path

ROOT = Path.home() / "KOBLLUX"
sys.path.append(str(ROOT / "python" / "lib"))
try:
    import veeb_runtime_gtts as runtime
except Exception as e:
    print("[erro] não achei veeb_runtime_gtts.py em python/lib. Detalhe:", e)
    sys.exit(2)

def read_text_from_file(path: Path) -> str:
    return path.read_text(encoding="utf-8")

def main(argv=None) -> int:
    p = argparse.ArgumentParser(description="KOBLLUX TTS CLI (gTTS runtime)")
    g = p.add_mutually_exclusive_group(required=True)
    g.add_argument("--text", help="Texto direto para leitura.")
    g.add_argument("--file", help="Caminho de arquivo com o texto a ser lido.")
    p.add_argument("--archetype", "-a", default="NOVA", help="Arquétipo (ex.: NOVA, HÓRUS, TRINITY...).")
    p.add_argument("--engine", "-e", default=None, help="Override de engine (ex.: gtts).")
    p.add_argument("--no-play", action="store_true", help="Não tocar após gerar.")
    p.add_argument("--cfg", default=str(ROOT / "configs" / "voices.json"), help="Caminho para voices.json.")
    p.add_argument("--out-root", default=str(ROOT / "out" / "audio"), help="Raiz de saída para áudios.")
    p.add_argument("--events-root", default=str(ROOT / "data" / "events"), help="Raiz para eventos.")
    args = p.parse_args(argv)

    if args.file:
        texto = read_text_from_file(Path(args.file))
    else:
        texto = args.text

    meta = runtime.speak(
        texto=texto,
        archetype=args.archetype,
        engine_override=args.engine,
        cfg_path=Path(args.cfg),
        out_root=Path(args.out_root),
        events_root=Path(args.events_root),
        play=(not args.no_play),
    )
    print(json.dumps(meta, ensure_ascii=False, indent=2))
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
